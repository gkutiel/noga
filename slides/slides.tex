\newcommand{\dcell}[4]{%
  \only<#1>{%
    \begin{scope}[shift={({#2*2.7},{-#3*2.05})}]
      \draw[gray!40,dashed] (1,0) -- (1,1.55);
      \draw[->,gray!65] (-0.15,0) -- (2.2,0);
      \draw[very thick,nogablue,line cap=round]
            plot[domain=0:2,samples=80,smooth] (\x,{#4});
    \end{scope}%
  }%
}
